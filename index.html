html

<!-- Контейнер, внутри которого можно двигать картинку -->

<div id="canvas-area" style="position: relative; width: 100%; height: 500px; border: 1px dashed #ccc; overflow: hidden;">

  

  <!-- Ваша картинка -->

  <img id="draggable-img" 

       src="https://sketchfab.com/3d-models/djingereyber-mosque-timbuktu-mali-cedd63bfa1e64779a950fd43b431a765" 

       alt="Дизайн-элемент" 

       style="position: absolute; top: 50px; left: 50px; width: 200px; cursor: grab; user-select: none; touch-action: none;">



</div>



<script>

  const img = document.getElementById('draggable-img');

  const container = document.getElementById('canvas-area');

  

  let isDragging = false;

  let startX, startY, initialLeft, initialTop;



  // Начало перетаскивания

  img.addEventListener('pointerdown', (e) => {

    isDragging = true;

    img.style.cursor = 'grabbing';

    

    // Запоминаем начальные координаты курсора и картинки

    startX = e.clientX;

    startY = e.clientY;

    initialLeft = img.offsetLeft;

    initialTop = img.offsetTop;

    

    // Захватываем указатель, чтобы картинка не терялась при резком движении

    img.setPointerCapture(e.pointerId);

  });



  // Процесс движения

  img.addEventListener('pointermove', (e) => {

    if (!isDragging) return;

    

    const dx = e.clientX - startX;

    const dy = e.clientY - startY;

    

    img.style.left = `${initialLeft + dx}px`;

    img.style.top = `${initialTop + dy}px`;

  });



  // Конец перетаскивания

  img.addEventListener('pointerup', (e) => {

    isDragging = false;

    img.style.cursor = 'grab';

    img.releasePointerCapture(e.pointerId);

  });

</script>
